Exercices Frontend Vue.js 3 - Intégration des utilisateurs ​

Ces exercices se concentrent sur l'intégration du modèle User avec le frontend Vue.js. Ils sont conçus pour être réalisés en environ 2 heures et s'appuient sur les connaissances déjà acquises avec le CRUD des tutoriels.

Exercice 1 : Création du service de données pour les utilisateurs ​

Objectif : Créer un service pour interagir avec l'API d'utilisateurs.

Durée estimée : 15-20 minutes

  1. Créez un fichier UserDataService.js dans le dossier services pour gérer les requêtes liées aux utilisateurs :
javascript
import http from "../http-common";

class UserDataService {
  getAll() {
    return http.get("/users");
  }

  get(id) {
    return http.get(`/users/${id}`);
  }

  create(data) {
    return http.post("/users", data);
  }

  update(id, data) {
    return http.put(`/users/${id}`, data);
  }

  delete(id) {
    return http.delete(`/users/${id}`);
  }

  findByUsername(username) {
    return http.get(`/users?username=${username}`);
  }
}

export default new UserDataService();

Exercice 2 : Création du composant de liste des utilisateurs ​

Objectif : Développer un composant pour afficher la liste des utilisateurs.

Durée estimée : 20-25 minutes

  1. Créez un composant UsersList.vue dans le dossier components :
vue
<template>
  <div class="list row">
    <div class="col-md-8">
      <div class="input-group mb-3">
        <input
          type="text"
          class="form-control"
          placeholder="Rechercher par nom d'utilisateur"
          v-model="username"
        />
        <div class="input-group-append">
          <button
            class="btn btn-outline-secondary"
            type="button"
            @click="searchUsername"
          >
            Rechercher
          </button>
        </div>
      </div>
    </div>

    <div class="col-md-6">
      <h4>Liste des utilisateurs</h4>
      <ul class="list-group">
        <li
          class="list-group-item"
          :class="{ active: index == currentIndex }"
          v-for="(user, index) in users"
          :key="index"
          @click="setActiveUser(user, index)"
        >
          {{ user.username }}
        </li>
      </ul>
    </div>
    
    <div class="col-md-6">
      <div v-if="currentUser">
        <h4>Utilisateur</h4>
        <div>
          <label><strong>Nom d'utilisateur:</strong></label> {{ currentUser.username }}
        </div>
        <div>
          <label><strong>Email:</strong></label> {{ currentUser.email }}
        </div>
        <div>
          <label><strong>Rôle:</strong></label> {{ currentUser.isAdmin ? "Administrateur" : "Utilisateur standard" }}
        </div>

        <router-link
          :to="'/users/' + currentUser.id"
          class="badge badge-warning"
        >
          Modifier
        </router-link>
      </div>
      <div v-else>
        <br />
        <p>Veuillez cliquer sur un utilisateur...</p>
      </div>
    </div>
  </div>
</template>

<script>
import UserDataService from "../services/UserDataService";

export default {
  name: "users-list",
  data() {
    return {
      users: [],
      currentUser: null,
      currentIndex: -1,
      username: ""
    };
  },
  methods: {
    retrieveUsers() {
      UserDataService.getAll()
        .then(response => {
          this.users = response.data;
          console.log(response.data);
        })
        .catch(e => {
          console.log(e);
        });
    },

    refreshList() {
      this.retrieveUsers();
      this.currentUser = null;
      this.currentIndex = -1;
    },

    setActiveUser(user, index) {
      this.currentUser = user;
      this.currentIndex = index;
    },
    
    searchUsername() {
      UserDataService.findByUsername(this.username)
        .then(response => {
          this.users = response.data;
          this.setActiveUser(null);
          console.log(response.data);
        })
        .catch(e => {
          console.log(e);
        });
    }
  },
  mounted() {
    this.retrieveUsers();
  }
};
</script>

Exercice 3 : Création du composant d'ajout d'utilisateur ​

Objectif : Développer un composant pour ajouter un nouvel utilisateur.

Durée estimée : 20-25 minutes

  1. Créez un composant AddUser.vue dans le dossier components :
vue
<template>
  <div class="submit-form">
    <div v-if="!submitted">
      <h4>Ajouter un utilisateur</h4>
      <div class="form-group">
        <label for="username">Nom d'utilisateur</label>
        <input
          type="text"
          class="form-control"
          id="username"
          required
          v-model="user.username"
          name="username"
        />
      </div>

      <div class="form-group">
        <label for="email">Email</label>
        <input
          type="email"
          class="form-control"
          id="email"
          required
          v-model="user.email"
          name="email"
        />
      </div>

      <div class="form-group">
        <label for="password">Mot de passe</label>
        <input
          type="password"
          class="form-control"
          id="password"
          required
          v-model="user.password"
          name="password"
        />
      </div>

      <div class="form-group form-check">
        <input
          type="checkbox"
          class="form-check-input"
          id="isAdmin"
          v-model="user.isAdmin"
          name="isAdmin"
        />
        <label class="form-check-label" for="isAdmin">Administrateur</label>
      </div>

      <button @click="saveUser" class="btn btn-success">Soumettre</button>
    </div>

    <div v-else>
      <h4>Utilisateur ajouté avec succès!</h4>
      <button class="btn btn-success" @click="newUser">Ajouter un autre</button>
    </div>
  </div>
</template>

<script>
import UserDataService from "../services/UserDataService";

export default {
  name: "add-user",
  data() {
    return {
      user: {
        id: null,
        username: "",
        email: "",
        password: "",
        isAdmin: false
      },
      submitted: false
    };
  },
  methods: {
    saveUser() {
      var data = {
        username: this.user.username,
        email: this.user.email,
        password: this.user.password,
        isAdmin: this.user.isAdmin
      };

      UserDataService.create(data)
        .then(response => {
          this.user.id = response.data.id;
          console.log(response.data);
          this.submitted = true;
        })
        .catch(e => {
          console.log(e);
        });
    },
    
    newUser() {
      this.submitted = false;
      this.user = {
        id: null,
        username: "",
        email: "",
        password: "",
        isAdmin: false
      };
    }
  }
};
</script>

Exercice 4 : Création du composant d'édition d'utilisateur ​

Objectif : Développer un composant pour afficher et modifier les détails d'un utilisateur.

Durée estimée : 20-25 minutes

  1. Créez un composant User.vue dans le dossier components :
vue
<template>
  <div v-if="currentUser" class="edit-form">
    <h4>Utilisateur</h4>
    <form>
      <div class="form-group">
        <label for="username">Nom d'utilisateur</label>
        <input
          type="text"
          class="form-control"
          id="username"
          v-model="currentUser.username"
        />
      </div>
      <div class="form-group">
        <label for="email">Email</label>
        <input
          type="email"
          class="form-control"
          id="email"
          v-model="currentUser.email"
        />
      </div>
      
      <div class="form-group">
        <label for="newPassword">Nouveau mot de passe (laisser vide pour ne pas changer)</label>
        <input
          type="password"
          class="form-control"
          id="newPassword"
          v-model="newPassword"
        />
      </div>

      <div class="form-group form-check">
        <input
          type="checkbox"
          class="form-check-input"
          id="isAdmin"
          v-model="currentUser.isAdmin"
        />
        <label class="form-check-label" for="isAdmin">Administrateur</label>
      </div>
    </form>

    <button class="badge badge-danger mr-2" @click="deleteUser">
      Supprimer
    </button>

    <button type="submit" class="badge badge-success" @click="updateUser">
      Mettre à jour
    </button>
    
    <div v-if="message" class="alert alert-success mt-3">
      {{ message }}
    </div>
  </div>

  <div v-else>
    <br />
    <p>Veuillez cliquer sur un utilisateur...</p>
  </div>
</template>

<script>
import UserDataService from "../services/UserDataService";

export default {
  name: "user",
  data() {
    return {
      currentUser: null,
      newPassword: "",
      message: ""
    };
  },
  methods: {
    getUser(id) {
      UserDataService.get(id)
        .then(response => {
          this.currentUser = response.data;
          console.log(response.data);
        })
        .catch(e => {
          console.log(e);
        });
    },

    updateUser() {
      const data = { ...this.currentUser };
      
      // Ajouter le nouveau mot de passe seulement s'il est fourni
      if (this.newPassword) {
        data.password = this.newPassword;
      }
      
      UserDataService.update(this.currentUser.id, data)
        .then(response => {
          console.log(response.data);
          this.message = "L'utilisateur a été mis à jour avec succès!";
          this.newPassword = ""; // Réinitialiser le champ du mot de passe
        })
        .catch(e => {
          console.log(e);
        });
    },

    deleteUser() {
      if (confirm("Êtes-vous sûr de vouloir supprimer cet utilisateur ?")) {
        UserDataService.delete(this.currentUser.id)
          .then(response => {
            console.log(response.data);
            this.$router.push({ name: "users" });
          })
          .catch(e => {
            console.log(e);
          });
      }
    }
  },
  mounted() {
    this.message = "";
    this.getUser(this.$route.params.id);
  }
};
</script>

Exercice 5 : Service d'authentification simplifié ​

Objectif : Créer un service d'authentification basique.

Durée estimée : 15-20 minutes

  1. Créez un fichier AuthService.js dans le dossier services :
javascript
import http from "../http-common";

class AuthService {
  login(credentials) {
    return http.post("/auth/signin", credentials)
      .then(response => {
        if (response.data && response.data.accessToken) {
          localStorage.setItem('user', JSON.stringify(response.data));
        }
        return response.data;
      });
  }

  logout() {
    localStorage.removeItem('user');
  }

  register(user) {
    return http.post("/auth/signup", user);
  }

  getCurrentUser() {
    return JSON.parse(localStorage.getItem('user'));
  }

  // Méthode utilitaire pour ajouter le token aux requêtes
  getAuthHeader() {
    const user = this.getCurrentUser();
    if (user && user.accessToken) {
      return { 'x-access-token': user.accessToken };
    } else {
      return {};
    }
  }
}

export default new AuthService();

Exercice 6 : Composant de connexion ​

Objectif : Créer un formulaire de connexion.

Durée estimée : 15-20 minutes

  1. Créez un composant Login.vue dans le dossier components :
vue
<template>
  <div class="col-md-12">
    <div class="card card-container">
      <img
        id="profile-img"
        src="//ssl.gstatic.com/accounts/ui/avatar_2x.png"
        class="profile-img-card"
        alt="Avatar"
      />
      <form name="form" @submit.prevent="handleLogin">
        <div class="form-group">
          <label for="username">Nom d'utilisateur</label>
          <input
            v-model="user.username"
            type="text"
            class="form-control"
            name="username"
          />
        </div>
        <div class="form-group">
          <label for="password">Mot de passe</label>
          <input
            v-model="user.password"
            type="password"
            class="form-control"
            name="password"
          />
        </div>
        <div class="form-group">
          <button class="btn btn-primary btn-block" :disabled="loading">
            <span v-show="loading" class="spinner-border spinner-border-sm"></span>
            <span>Connexion</span>
          </button>
        </div>
        <div class="form-group">
          <div v-if="message" class="alert alert-danger" role="alert">
            {{ message }}
          </div>
        </div>
      </form>
    </div>
  </div>
</template>

<script>
import AuthService from '../services/AuthService';

export default {
  name: 'login',
  data() {
    return {
      user: {
        username: '',
        password: ''
      },
      loading: false,
      message: ''
    };
  },
  methods: {
    handleLogin() {
      this.loading = true;
      this.message = '';
      
      AuthService.login(this.user)
        .then(
          response => {
            this.$router.push('/profile');
          },
          error => {
            this.loading = false;
            this.message = error.response.data.message || 
                          'Une erreur est survenue lors de la connexion.';
          }
        );
    }
  }
};
</script>

  1. Créez un composant UserTutorials.vue dans le dossier components :
vue
<template>
  <div class="list row">
    <div class="col-md-8">
      <h4>Tutoriels de l'utilisateur: {{ username }}</h4>
    </div>
    
    <div class="col-md-12">
      <ul class="list-group">
        <li
          class="list-group-item"
          v-for="(tutorial, index) in tutorials"
          :key="index"
        >
          <div>
            <strong>{{ tutorial.title }}</strong>
          </div>
          <div>
            {{ tutorial.description }}
          </div>
          <div>
            <small>Statut: {{ tutorial.published ? "Publié" : "En attente" }}</small>
          </div>
          <div>
            <router-link
              :to="'/tutorials/' + tutorial.id"
              class="badge badge-warning"
            >
              Modifier
            </router-link>
          </div>
        </li>
      </ul>

      <div v-if="tutorials.length === 0" class="alert alert-info mt-3">
        Cet utilisateur n'a pas de tutoriels.
      </div>

      <router-link to="/users" class="btn btn-primary mt-3">
        Retour à la liste des utilisateurs
      </router-link>
    </div>
  </div>
</template>

<script>
import TutorialDataService from "../services/TutorialDataService";
import UserDataService from "../services/UserDataService";

export default {
  name: "user-tutorials",
  data() {
    return {
      tutorials: [],
      username: "",
      userId: null
    };
  },
  methods: {
    retrieveTutorials() {
      TutorialDataService.findByUser(this.userId)
        .then(response => {
          this.tutorials = response.data;
          console.log(response.data);
        })
        .catch(e => {
          console.log(e);
        });
    },

    getUsername() {
      UserDataService.get(this.userId)
        .then(response => {
          this.username = response.data.username;
        })
        .catch(e => {
          console.log(e);
        });
    }
  },
  mounted() {
    this.userId = this.$route.params.id;
    this.getUsername();
    this.retrieveTutorials();
  }
};
</script>

  1. Ajoutez la route correspondante dans votre fichier de routage.

Configuration complète des routes ​

Pour que tous ces composants fonctionnent ensemble, assurez-vous que votre fichier router.js contient ces routes :

javascript
import { createRouter, createWebHistory } from 'vue-router'

const routes = [
  {
    path: "/",
    alias: "/tutorials",
    name: "tutorials",
    component: () => import("./components/TutorialsList")
  },
  {
    path: "/tutorials/:id",
    name: "tutorial-details",
    component: () => import("./components/Tutorial")
  },
  {
    path: "/add",
    name: "add",
    component: () => import("./components/AddTutorial")
  },
  // Routes pour les utilisateurs
  {
    path: "/users",
    name: "users",
    component: () => import("./components/UsersList")
  },
  {
    path: "/users/:id",
    name: "user-details",
    component: () => import("./components/User")
  },
  {
    path: "/users/add",
    name: "add-user",
    component: () => import("./components/AddUser")
  },
  {
    path: "/users/:id/tutorials",
    name: "user-tutorials",
    component: () => import("./components/UserTutorials")
  },
  // Routes pour l'authentification
  {
    path: "/login",
    name: "login",
    component: () => import("./components/Login")
  }
]

const router = createRouter({
  history: createWebHistory(),
  routes
})

export default router
